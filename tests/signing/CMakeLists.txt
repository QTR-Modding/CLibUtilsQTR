cmake_minimum_required(VERSION 3.25)
project(QTRSigningTests LANGUAGES CXX)
enable_testing()
add_subdirectory("../../tools/signing" "tools")
foreach(kind IN ITEMS signed unsigned wrong)
    add_library(${kind}_provider SHARED Provider.cpp)
    set_target_properties(${kind}_provider PROPERTIES OUTPUT_NAME ExampleProvider
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${kind}")
endforeach()
add_executable(binding_tests BindingTests.cpp)
add_executable(signature_parameter_tests SignatureParameterTests.cpp)
target_compile_features(signature_parameter_tests PRIVATE cxx_std_23)
target_include_directories(signature_parameter_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
if(MSVC)
    target_compile_options(signature_parameter_tests PRIVATE /W4 /WX /permissive- /EHsc)
endif()
add_test(NAME signing.parameter_diagnostics COMMAND signature_parameter_tests)
add_executable(digest_tests DigestTests.cpp)
target_compile_features(digest_tests PRIVATE cxx_std_23)
target_include_directories(digest_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include" "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(digest_tests PRIVATE /W4 /WX /permissive- /EHsc)
endif()
add_test(NAME signing.digest COMMAND digest_tests "$<TARGET_FILE:signed_provider>" portable)
add_test(NAME signing.windows_digest COMMAND digest_tests "$<TARGET_FILE:signed_provider>" oracle)
target_compile_features(binding_tests PRIVATE cxx_std_23)
target_include_directories(binding_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include" "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(binding_tests PRIVATE /W4 /WX /permissive- /EHsc)
endif()
foreach(mode IN ITEMS valid retry code exports)
    add_test(NAME signing.${mode} COMMAND binding_tests ${mode} "$<TARGET_FILE:signed_provider>")
endforeach()
foreach(kind IN ITEMS unsigned wrong)
    add_test(NAME signing.${kind} COMMAND binding_tests reject "$<TARGET_FILE:${kind}_provider>")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/artifacts-$<CONFIG>.json" CONTENT
"{\"signed\":\"$<TARGET_FILE:signed_provider>\",\"wrong\":\"$<TARGET_FILE:wrong_provider>\",\"verifier\":\"$<TARGET_FILE:qtr_verify_signature>\"}")
